import { Meta } from '@storybook/addon-docs/blocks';
import data from '../docs/core-theme-issues.json';

export const groups = [
    { status: 'solved', label: 'Solved' },
    { status: 'partly', label: 'Partly solved' },
    { status: 'check', label: 'To check' },
    { status: 'irrelevant', label: 'Not relevant here' },
];
export const itemsOf = (status) => data.items.filter((item) => item.status === status);

<Meta title="Core theme issues" tags={['dev']} />

# Lessons from the core theme

<p className="issue-lede" data-theme="mono">
    {
        "Every problem reported for TYPO3's core theme Camino, and what Kumano Kodo does about it. Each row is one problem and its solution here; the numbers link to the reports."
    }
</p>

<ul className="issue-summary" data-theme="mono">
    {groups
        .filter((group) => itemsOf(group.status).length)
        .map((group) => (
            <li key={group.status} className={`issue-summary-item issue-${group.status}`}>
                <strong>{itemsOf(group.status).length}</strong> {group.label}
            </li>
        ))}
</ul>

{groups.filter((group) => itemsOf(group.status).length).map((group) => (

<section key={group.status} className={`issue-group issue-${group.status}`} data-theme="mono">
<h2 className="issue-group-title">{group.label}</h2>
<ul className="issue-list">
<li className="issue issue-head" aria-hidden="true">
<p className="issue-problem">Problem</p>
<p className="issue-solution">Solution here</p>
<p className="issue-refs">Reports</p>
</li>
{itemsOf(group.status).map((item) => (
<li key={item.problem} className="issue">
<p className="issue-problem">{item.problem}</p>
<p className="issue-solution">{item.solution}</p>
<p className="issue-refs">
{item.issues.length
? item.issues.map((n) => (
<a key={n} href={`https://forge.typo3.org/issues/${n}`} target="\_blank" rel="noreferrer">
#{n}
</a>
))
: 'Core team feedback'}
</p>
</li>
))}
</ul>
</section>
))}

<p className="issue-source" data-theme="mono">
    {data.source}
</p>
