import { Meta } from '@storybook/addon-docs/blocks';
import data from '../docs/feedback.json';

export const groups = [
    { status: 'solved', label: 'Done' },
    { status: 'partly', label: 'Partly done' },
    { status: 'check', label: 'Open' },
    { status: 'irrelevant', label: 'Deferred' },
];
export const itemsOf = (status) => data.items.filter((item) => item.status === status);
export const hrefOf = (link) =>
    link.commit
        ? `https://github.com/hahnmg/kumano-kodo/commit/${link.commit}`
        : `./?path=/${link.docs ? 'docs' : 'story'}/${link.docs ?? link.story}`;

<Meta title="Feedback" tags={['dev']} />

# Feedback and what changed

<p className="issue-lede" data-theme="mono">
    {
        'The feedback on Kumano Kodo for TYPO3 so far, and what changed for each point. Names link to the result in this Storybook, hashes to the commit.'
    }
</p>

<ul className="issue-summary" data-theme="mono">
    {groups
        .filter((group) => itemsOf(group.status).length)
        .map((group) => (
            <li key={group.status} className={`issue-summary-item issue-${group.status}`}>
                <strong>{itemsOf(group.status).length}</strong> {group.label}
            </li>
        ))}
</ul>

{groups.filter((group) => itemsOf(group.status).length).map((group) => (

<section key={group.status} className={`issue-group issue-${group.status}`} data-theme="mono">
<h2 className="issue-group-title">{group.label}</h2>
<ul className="issue-list">
<li className="issue issue-head" aria-hidden="true">
<p className="issue-problem">Feedback</p>
<p className="issue-solution">What changed</p>
<p className="issue-refs">See</p>
</li>
{itemsOf(group.status).map((item) => (
<li key={item.feedback} className="issue">
<p className="issue-problem">
<strong>{item.from}:</strong> {item.feedback}
</p>
<p className="issue-solution">{item.change}</p>
<p className="issue-refs">
{item.links.length
? item.links.map((link) => (
<a key={link.commit ?? link.label} href={hrefOf(link)} target={link.commit ? '\_blank' : '\_top'} rel="noreferrer">
{link.commit ?? link.label}
</a>
))
: '–'}
</p>
</li>
))}
</ul>
</section>
))}

<p className="issue-source" data-theme="mono">
    {data.source}
</p>
